        function updateLabel(index) {

            const data = projectData[index];

            const body = document.getElementById('main-body');

            const profileLink = document.getElementById('profile-link');

            

            body.classList.remove('profile-active', 'force-white');


            if (data.isProfile) {

                body.classList.add('profile-active');

                profileLink.style.display = 'none';

                document.getElementById('label-container').innerHTML = `

                    <div style="max-width: 420px; text-transform: none; text-align: left;">${data.content}</div>

                `;

            } else {

                if (index === 3) {

                    body.classList.add('force-white');

                }

                

                profileLink.style.display = 'block';

                document.getElementById('label-container').innerHTML = `

                    <div>PRODUCT: ${data.name}</div>

                    <div>TYPE:    ${data.type}</div>

                    <div>TEAM:    ${data.team}</div>

                    <div>STATUS:  ${data.status}</div>

                `;

            }


            const pageNum = (index + 1).toString().padStart(2, '0');

            const totalNum = sections.length.toString().padStart(2, '0');

            document.getElementById('page-counter').innerText = `${pageNum} / ${totalNum}`;

        }


        async function playVideo(index) {

            sections.forEach(s => {

                const v = s.querySelector('video');

                if (v) {

                    v.pause();

                    v.classList.remove('visible');

                    v.currentTime = 0;

                }

            });


            const activeVideo = sections[index].querySelector('video');

            if (activeVideo) {

                try {

                    await activeVideo.play();

                    activeVideo.classList.add('visible');

                } catch (e) {}

            }

        }


        let isLocked = false;

        function transitionTo(index) {

            if (isLocked || index < 0 || index >= sections.length || index === currentIndex) return;

            isLocked = true;


            const prevIndex = currentIndex;

            sections[prevIndex].classList.replace('active', 'inactive');

            sections[index].classList.replace('inactive', 'active');


            currentIndex = index;

            updateLabel(index);

            playVideo(index);


            setTimeout(() => {

                isLocked = false;

            }, 800);

        }


        document.getElementById('profile-link').addEventListener('click', () => {

            transitionTo(sections.length - 1);

        });


        sections.forEach((s, i) => {

            if (i !== 0) s.classList.add('inactive');

        });


        updateLabel(0);

        window.addEventListener('load', () => playVideo(0));

        window.addEventListener('wheel', e => {

            if (Math.abs(e.deltaY) < 30) return;

            transitionTo(e.deltaY > 0 ? currentIndex + 1 : currentIndex - 1);

        }, { passive: true });


        let startY = 0;

        window.addEventListener('touchstart', e => startY = e.touches[0].clientY, { passive: true });

        window.addEventListener('touchend', e => {

            const diff = startY - e.changedTouches[0].clientY;

            if (Math.abs(diff) > 50) transitionTo(diff > 0 ? currentIndex + 1 : currentIndex - 1);

        }, { passive: true });


        document.addEventListener('click', () => playVideo(currentIndex), { once: true });
